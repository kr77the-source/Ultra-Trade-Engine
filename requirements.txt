import yfinance as yf
import pandas as pd
import config

class LiveDataManager:
    def __init__(self):
        # Symbol mapping for real data provider (e.g., Nifty 50 is "^NSEI")
        self.symbol = "^NSEI" if "NIFTY" in config.DEFAULT_SYMBOL.upper() else config.DEFAULT_SYMBOL
        
    def get_latest_price(self):
        """Fetches the absolute latest real market price from live feed."""
        ticker = yf.Ticker(self.symbol)
        # Fetch fast live info or latest intraday data
        todays_data = ticker.history(period="1d", interval="1m")
        if todays_data.empty:
            raise ValueError(f"Could not fetch real live price for {self.symbol}. Check market hours or network.")
        
        latest_price = float(todays_data['Close'].iloc[-1])
        return round(latest_price, 2)

    def get_historical_data(self, interval="15m", period="30d"):
        """Fetches real historical candles directly from live market feeds."""
        # Convert config intervals to yfinance compatible intervals
        yf_interval_map = {
            "1m": "1m",
            "5m": "5m",
            "15m": "15m",
            "30m": "30m",
            "1h": "60m",
            "1D": "1d"
        }
        yf_interval = yf_interval_map.get(interval, "15m")
        
        ticker = yf.Ticker(self.symbol)
        df = ticker.history(period=period, interval=yf_interval)
        
        if df.empty:
            raise ValueError(f"Failed to fetch real historical data for {self.symbol} with period {period}.")
            
        # Keep only required columns
        df = df[['Open', 'High', 'Low', 'Close', 'Volume']]
        return df

def fetch_live_feed():
    manager = LiveDataManager()
    return manager.get_latest_price()
